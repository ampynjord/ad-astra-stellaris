# Ad Astra - surcharge de l'evenement vanilla action.62 (« End of the X »),
# copie exacte 4.5.3 (events/on_action_events_1.txt) + un trigger marque Ad Astra.
#
# 11/11 - partie de test beta.12 (Early Space, confinee) : « End of the Ymaceran
# Mandate » nommait une nation stellaire detruite avant l'emergence. Le jeu de
# base diffuse l'evenement a tout pays jouable en communication avec le pays
# detruit ; la veille du mod ne coupe pas ces communications-la. Toute mise a
# jour du jeu de base doit etre repercutee ici.

namespace = action

country_event = {
	id = action.62
	title = "action.62.name"
	desc = "action.62.desc"
	show_sound = event_default
	picture = "GFX_evt_city_ruins"
	is_triggered_only = yes

	# Ad Astra : un empire confine ne sait pas qu'une nation stellaire existe, encore
	# moins qu'elle vient d'etre detruite (cartes B20/B25, constate en jeu 2240.04).
	trigger = { core_est_confine = no }

	option = {
		name = OK
	}

	option = {
		name = "action.62.memorialist.a"
		exclusive_trigger = {
			is_memorialist_empire = yes
		}
	}

	after = {
		if = {
			limit = {
				has_country_flag = payback_stolen_revenge
			}
			country_event = { id = origin.4070 days = 15 }
		}
		else_if = {
			limit = {
				has_country_flag = payback_killed_MSI
			}
			country_event = { id = origin.4061 }
		}
	}
}
