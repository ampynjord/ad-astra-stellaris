# common/on_actions/adastra_outliner_on_actions.txt. Il ne modifie aucun
# evenement existant.

# 23/09 - test en 4.5 (partie Sol, 2201.02.12) : l'empire confine n'a AUCUN
# secteur (sectors vide dans la save, ADASTRA DIAG « capitale SANS secteur »)
# et Terra reste pourtant listee, sous « Secteur frontalier ». Le panneau
# n'exige donc pas de secteur : la premiere hypothese (planete absente faute
# de secteur a nous) est fausse.
#
# 12/11 - cause PROUVEE au banc (carte B22) : quand un autre empire pose son
# avant-poste chez nous, le systeme rejoint SON secteur, notre planete le
# suit et sort de notre groupe frontalier. Limite moteur : create_sector ne
# peut rien tant que la base n'est pas a nous.
#
# 13/11 - les lignes ADASTRA DIAG de game.log, qui ont servi a cette preuve,
# sont retirees. adastra.176 ne fait plus que prevenir le joueur, une fois
# par occupation (adastra.177), avec le bouton pour aller a sa planete.

# adastra.176 - veille du panneau lateral (mensuel, cache)
country_event = {
	id = adastra.176
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_adastra
		NOT = { has_country_flag = adastra_completed }
		exists = capital_scope
	}

	immediate = {
		# Aide au joueur : le systeme natal est a un autre empire. Pas de
		# condition sur le secteur (23/09) : la capitale n'en a jamais, et sous
		# occupation elle peut tomber dans celui de l'occupant. Une fenetre par
		# occupation ; le drapeau tombe des que le systeme n'est plus a un
		# autre, pour qu'une nouvelle occupation previenne a nouveau.
		if = {
			limit = {
				capital_scope = {
					solar_system = {
						has_owner = yes
						NOT = { is_owned_by = root }
					}
				}
			}
			if = {
				limit = { NOT = { has_country_flag = adastra_panneau_prevenu } }
				set_country_flag = adastra_panneau_prevenu
				capital_scope = { save_event_target_as = adastra_monde_natal }
				country_event = { id = adastra.177 }
			}
		}
		else_if = {
			limit = { has_country_flag = adastra_panneau_prevenu }
			remove_country_flag = adastra_panneau_prevenu
		}
	}
}

	option = {
		name = "adastra.177.a"
		custom_tooltip = "adastra.177.recap"
	}
}
