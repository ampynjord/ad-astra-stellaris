# Ad Astra 1.4 (integration pre-PRL) - 2026-08-28
#
# Fichier AUTONOME, a retirer d'un bloc pour revenir a la beta.4.
# Il ne modifie aucun evenement existant : tout passe par de nouveaux
# evenements branches sur le pouls mensuel via un fichier d'on_action
# separe. Aucun fichier du lot de travail n'est touche.
#
# COUCHE A - synchronisation de l'age pre-PRL moteur.
# Ad Astra represente ses ages par des drapeaux `adastra_reached_*` qui
# s'accumulent. Le moteur, lui, a sa propre notion d'age pre-PRL, lue par
# `has_pre_ftl_age` (96 usages vanilla : designations de colonie, armees
# d'epoque, regroupements par ere). On aligne la seconde sur la premiere.
#
# ON N'APPELLE PAS `set_pre_ftl_age_effect` : cet effet scripte du vanilla
# fait `capital_scope = { remove_all_buildings = yes }` a chaque montee
# d'age, change le gouvernement en `authority = random` et demarre la
# situation de progression vanilla. Sur un empire joueur ce serait
# devastateur. On utilise l'effet moteur nu `set_pre_ftl_age`, qui ne fait
# que poser la propriete.

# (integre au namespace adastra, ids 160-169)

# adastra.160 - aligne l'age moteur sur l'age Ad Astra (mensuel, idempotent)
country_event = {
	id = adastra.160
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_adastra
		NOT = { has_country_flag = adastra_completed }
	}

	immediate = {
		# 30/08 : la logique vit dans adastra_preftl_effects.txt, partagee
		# avec adastra.2 (appel immediat au demarrage) - retour beta.5 : la
		# designation restait invisible tout le premier mois.
		adastra_preftl_sync = yes
		adastra_maj_designation = yes

		# 12/09 - LE SECTEUR EXIGE LA BASE STELLAIRE. Le diagnostic pose le
		# 31/08 a rendu son verdict dans le game.log d'argroww (beta.7,
		# carte B22) : 435 tentatives, 435 refus, zero reussite, et le
		# moteur donne sa raison dans error.log -
		#
		#   effect_impl_colony.cpp:113 : create_sector effect failed with
		#   reasons: We do not own the starbase in <systeme>
		#
		# Un empire confine n'a pas de base stellaire : l'appel etait donc
		# voue a l'echec a chaque mois, et ne produisait que du bruit de
		# journal. On ne tente plus la creation que lorsqu'elle peut
		# aboutir, c'est-a-dire quand la base du systeme natal est NOTRE :
		# apres la cession (adastra_cession_du_ciel) ou apres l'avoir
		# construite. La disparition de la planete du panneau lateral a
		# donc une AUTRE cause, encore inconnue : la planete est visible au
		# jour 1 alors qu'aucun secteur n'a jamais existe.
		if = {
			limit = {
				NOT = { num_sectors >= 1 }
				capital_scope = {
					solar_system = {
						exists = starbase
						starbase = { is_owned_by = root }
					}
				}
			}
			capital_scope = { create_sector = yes }
		}

		# 30/08 - L'ILLUMINATION RECUE. Le pacte vanilla pose son drapeau
		# sur nous ; on le traduit en effet reel : +25 % de vitesse de
		# recherche, -10 % d'unite (apprendre d'un maitre divise). Une
		# fenetre unique raconte le debut.
		if = {
			limit = {
				has_country_flag = open_technological_enlightenment_pact
				NOT = { has_modifier = adastra_illumination }
			}
			add_modifier = { modifier = adastra_illumination }
			if = {
				limit = { NOT = { has_country_flag = adastra_illumination_annoncee } }
				set_country_flag = adastra_illumination_annoncee
				country_event = { id = adastra.172 }
			}
		}
		else_if = {
			limit = {
				has_modifier = adastra_illumination
				NOT = { has_country_flag = open_technological_enlightenment_pact }
			}
			remove_modifier = adastra_illumination
		}
		# 1.4 (2026-08-29, revu 31/08) - tant que l'empire est confine, les
		# chaines de premier contact (les notres comme celles qui nous
		# visent) avancent au quart de leur vitesse : on ne « decouvre »
		# pas un empire stellaire depuis le sol en quelques semaines.
		# Save 2229.06.14 : la version a palier d'eveil tombait des 80
		# d'eveil et le contact vers l'occupant s'est plie en une etape -
		# le modificateur ne tombe plus qu'a l'emergence.
		if = {
			limit = {
				has_country_flag = adastra_locked
				NOT = { has_modifier = adastra_ciel_lointain }
			}
			add_modifier = { modifier = adastra_ciel_lointain }
		}
		else_if = {
			limit = {
				has_modifier = adastra_ciel_lointain
				NOT = { has_country_flag = adastra_locked }
			}
			remove_modifier = adastra_ciel_lointain
		}
	}
}

# COUCHE B - les sequelles d'epoque du vanilla.
#
# A partir de l'age machine, le vanilla pose deux blocages sur le monde
# d'une civilisation pre-PRL : logements decrepits et infrastructure
# defaillante. Chacun retire un district maximum et coute 300 energie a
# deblayer ; les logements decrepits *creent une population* quand on les
# demolit. C'est une mecanique d'urbanisation deja ecrite, et ces deux
# depots n'ont aucune garde `is_primitive` : ils fonctionnent tels quels.
#
# Une seule pose, definitive, a l'entree dans l'age machine.
country_event = {
	id = adastra.161
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_adastra
		NOT = { has_country_flag = adastra_completed }
		has_country_flag = adastra_reached_machine
		NOT = { has_country_flag = adastra_sequelles_posees }
	}

	immediate = {
		set_country_flag = adastra_sequelles_posees
		# 30/08 : l'initialiseur Sol du joueur pose deja ces blockers sur la
		# Terre. Sans garde, on les DOUBLAIT a l'age machine (retour beta.5).
		capital_scope = {
			if = {
				limit = { NOT = { has_deposit = d_decrepit_dwellings } }
				add_deposit = d_decrepit_dwellings
			}
			if = {
				limit = { NOT = { has_deposit = d_failing_infrastructure } }
				add_deposit = d_failing_infrastructure
			}
		}
	}
}

	}

# ============================================================================
# COUCHE H - LA NEGOCIATION DU CIEL (cohabitation, 30/08 ; refontes 31/08).
#
# Refonte finale 31/08 (demande d'ampynjord) : UN SEUL bouton diplomatique
# « Reclamer notre ciel » sur l'empire occupant, toujours disponible
# (common/diplomatic_actions/adastra_actions.txt, auto-accepte). Ce n'est
# pas un tirage : l'ETHIQUE de l'occupant tranche, deterministe.
#
# adastra.180 : le routeur. Sur l'ethique de l'occupant :
#     xenophobe            -> 187 refus + une REVENDICATION sur le systeme
#                             natal (la guerre devient la voie).
#     pacifiste / xenophile -> 181 cession gratuite.
#     autoritaire           -> 186 cession contre TRIBUT (devenir tributaire).
#     megacorp / materialiste / egalitaire -> 182 cession contre ressources.
#     militariste           -> 187 refus + revendication.
#     sinon                 -> 182 (l'interet, valeur par defaut).
# Chaque issue retrouve l'occupant via la base (encore a lui a cet instant).
# Cooldown 720 jours (adastra_negociation_recente) pour eviter le spam.
# ============================================================================
country_event = {
	id = adastra.180
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		set_timed_country_flag = { flag = adastra_negociation_recente days = 720 }
		capital_scope = {
			solar_system = {
				starbase = {
					owner = { save_event_target_as = adastra_occupant }
				}
			}
		}
		if = {
			limit = { event_target:adastra_occupant = { is_xenophobe = yes } }
			country_event = { id = adastra.187 }
		}
		else_if = {
			limit = {
				event_target:adastra_occupant = {
					OR = { is_pacifist = yes  is_xenophile = yes }
				}
			}
			country_event = { id = adastra.181 }
		}
		else_if = {
			limit = { event_target:adastra_occupant = { is_authoritarian = yes } }
			country_event = { id = adastra.186 }
		}
		else_if = {
			limit = {
				event_target:adastra_occupant = {
					OR = { is_megacorp = yes  is_materialist = yes  is_egalitarian = yes }
				}
			}
			country_event = { id = adastra.182 }
		}
		else_if = {
			limit = { event_target:adastra_occupant = { is_militarist = yes } }
			country_event = { id = adastra.187 }
		}
		else = {
			country_event = { id = adastra.182 }
		}
	}
}
