# Ad Astra 1.4 - les evenements des Premiers pas.
#
# adastra.100  lancement suborbital : on tire le sort
# adastra.101  ... et il a rate
# adastra.102  satellite en orbite
# adastra.104  vol habite
# adastra.106  premier corps atteint
# adastra.107  ... mais le systeme n'offre rien a atteindre
#
# LE SORT DU LANCEMENT. La probabilite d'echec part de 50 % et tombe a 10 %
# quand tout ce qui aide est connu. Ce qui aide : le bouclier ablatif, la
# theorie de l'information (le guidage), l'electronique. Le joueur peut donc
# choisir - tirer tout de suite et risquer, ou attendre trois technologies.
# C'est ca, un echec visiblement evitable.

namespace = adastra

# --------------------------------------------------------- 100 : on tire
country_event = {
	id = adastra.100
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		# On compte ce qui aide. Chaque element retire 10 points de risque.
		set_variable = { which = core_risque value = 50 }
		if = {
			limit = { has_technology = tech_adastra_heat_shield }
			change_variable = { which = core_risque value = -10 }
		}
		if = {
			limit = { has_technology = tech_adastra_information_theory }
			change_variable = { which = core_risque value = -10 }
		}
		if = {
			limit = { has_technology = tech_adastra_electronics }
			change_variable = { which = core_risque value = -10 }
		}
		# Un programme deja rode rate moins. La quatrieme tentative est plus
		# sure que la premiere, meme sans technologie nouvelle : c'est ce que
		# raconte l'histoire de tous les programmes spatiaux.
		if = {
			limit = { check_variable = { which = core_tirs value > 2 } }
			change_variable = { which = core_risque value = -10 }
		}
		change_variable = { which = core_tirs value = 1 }

		random_list = {
			50 = {
				modifier = {
					factor = 0
					check_variable = { which = core_risque value < 15 }
				}
				# ECHEC. On a paye, la fusee est perdue, et le pays le sait.
				country_event = { id = adastra.101 }
			}
			50 = {
				# REUSSITE.
				set_country_flag = core_suborbital_fait
				add_monthly_resource_mult = {
					resource = physics_research
					value = 3
					min = 30
					max = 300
				}
				add_modifier = {
					modifier = core_mod_suborbital
					days = -1
				}
				country_event = { id = adastra.103 }
			}
		}
	}
}

# ---------------------------------------------------- 106 : le premier corps
# Rien n'est nomme. On cherche ce que le systeme natal contient reellement,
# du plus evocateur au moins : une lune, puis un asteroide, puis n'importe
# quel astre qui n'est ni l'etoile ni le monde natal.
country_event = {
	id = adastra.106
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		set_country_flag = core_premier_corps_fait
		capital_scope = {
			save_event_target_as = core_natal
			solar_system = {
				# 1 - une lune
				random_system_planet = {
					limit = {
						is_moon = yes
						NOT = { is_same_value = event_target:core_natal }
					}
					save_global_event_target_as = core_cible
				}
				# 2 - sinon un asteroide
				if = {
					limit = { NOT = { exists = event_target:core_cible } }
					random_system_planet = {
						limit = {
							is_asteroid = yes
							NOT = { is_same_value = event_target:core_natal }
						}
						save_global_event_target_as = core_cible
					}
				}
				# 3 - sinon n'importe quel astre qui n'est ni l'etoile ni chez nous
				if = {
					limit = { NOT = { exists = event_target:core_cible } }
					random_system_planet = {
						limit = {
							is_star = no
							NOT = { is_same_value = event_target:core_natal }
						}
						save_global_event_target_as = core_cible
					}
				}
			}
		}
		if = {
			limit = { exists = event_target:core_cible }
			country_event = { id = adastra.108 }
		}
		else = {
			# Un systeme natal sans autre astre. C'est rare et c'est une
			# histoire en soi : on ne se pose nulle part, on tourne.
			country_event = { id = adastra.107 }
		}
	}
}

# ------------------------------------------------- 110 : la sonde arrive
# On cherche un systeme voisin dont l'etoile n'est pas encore connue. Si tout
# ce qui est a portee l'est deja, la sonde part quand meme - et rapporte de la
# science plutot que de la carte. Une decision ne doit jamais ne rien faire.
country_event = {
	id = adastra.110
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		# `random_system` parcourt la galaxie ; la limite de distance la ramene
		# au voisinage. On evite `random_system_within_border` : un empire
		# confine n'a de frontieres que son propre systeme, la portee serait
		# vide. On evite aussi de viser un systeme deja visite par une sonde.
		capital_scope = { solar_system = { save_event_target_as = core_sonde_source } }
		random_system = {
			limit = {
				NOT = { is_same_value = event_target:core_sonde_source }
				NOT = { has_star_flag = core_sonde_visitee }
				# 1.4 (17/08) : `max_distance` n'existe pas dans le declencheur
				# `distance` (qui ne connait que min_jumps/max_jumps, comme le
				# reste des evenements vanilla) - le parseur rejetait la ligne
				# ("Not a valid compare token '=' ... line : 288") et la
				# recherche de systeme voisin echouait silencieusement.
				distance = {
					source = event_target:core_sonde_source
					max_jumps = 4
				}
			}
			save_global_event_target_as = core_sonde_cible
		}
		if = {
			limit = { exists = event_target:core_sonde_cible }
			event_target:core_sonde_cible = {
				set_star_flag = core_sonde_visitee
				every_system_planet = {
					limit = { is_star = yes }
					set_surveyed = { surveyed = yes surveyor = root }
				}
			}
			country_event = { id = adastra.111 }
		}
		else = {
			country_event = { id = adastra.112 }
		}
	}
}
