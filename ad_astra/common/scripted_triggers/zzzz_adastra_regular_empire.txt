is_regular_empire = {
	OR = {
		is_country_type = default
		#for checks in buildings and districts
		is_country_type = awakened_marauders
		is_country_type = fallen_empire
		is_country_type = awakened_fallen_empire
		is_country_type = exiled
		is_country_type = adastra_grounded
		# Gigastructural Engineering & More : l'Imperium Katzenartig
		AND = {
			has_global_flag = katzenartig_imperium_happened
			NOT = { has_global_flag = katzenartig_imperium_defeated }
			exists = event_target:flusion_katzens
			NOT = { is_country_type = primitive }
			founder_species = { is_same_species = event_target:flusion_katzens }
		}
	}
	NOT = { has_ethic = ethic_gestalt_consciousness	}
}
