# Ad Astra 1.2 - verrous d'age.
# GENERE PAR tools/gen_age_techs.py - NE PAS EDITER A LA MAIN.
#
# adastra_age_<age>_done      : 6 des 10 technos de l'age sont terminees.
#   Un seuil, pas la totalite : un age propose 10 technos pour un budget
#   de recherche d'environ 8, donc exiger les 10 verrouillerait le vanilla
#   de l'age pour un joueur qui a simplement fait d'autres choix.
# adastra_vanilla_open_<age>  : garde utilisee par les overrides de techs
#   vanilla. Le vanilla d'un age ne s'ouvre qu'apres ses techs custom :
#   les techs custom PRECEDENT le vanilla, elles ne le concurrencent pas.
#   Le OR sur adastra_completed est un filet de securite : apres l'emergence
#   les techs d'age quittent le tirage, le vanilla ne doit jamais rester
#   verrouille a cause d'un age laisse inacheve.

# stone : 6 technos sur 10 suffisent.
adastra_age_stone_done = {
	calc_true_if = {
		amount >= 6
		has_technology = tech_adastra_fire
		has_technology = tech_adastra_stone_tools
		has_technology = tech_adastra_hunting
		has_technology = tech_adastra_cave_art
		has_technology = tech_adastra_language
		has_technology = tech_adastra_seasons

		has_technology = tech_adastra_cordage
		has_technology = tech_adastra_shelter
		has_technology = tech_adastra_burial
	}
}

# --- Ressources -------------------------------------------------------
# Une ressource n'existe pas avant l'invention qui la produit.
#
# La condition est la TECHNOLOGIE seule. C'est suffisant depuis que les ages
# deja traverses offrent leurs technologies au demarrage (adastra.2 appelle
# adastra_grant_age_<age> pour chaque age anterieur) : un empire qui commence
# a l'Age de l'atome connait deja le Reseau electrique, il ne l'a pas
# recherche mais il l'a. Un empire qui commence a l'Age de la machine, lui,
# doit encore l'inventer - c'est justement l'age ou il l'invente.

adastra_has_alloys = {
	OR = {
		has_country_flag = adastra_completed
		has_technology = tech_adastra_bronze
	}
}

adastra_has_consumer_goods = {
	OR = {
		has_country_flag = adastra_completed
		has_technology = tech_adastra_steam_engine
	}
}

adastra_has_energy = {
	OR = {
		has_country_flag = adastra_completed
		has_technology = tech_adastra_electricity
	}
}
