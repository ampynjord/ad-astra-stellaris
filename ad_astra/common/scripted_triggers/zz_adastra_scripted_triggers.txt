# Ad Astra - surcharge du déclencheur scripté vanilla is_regular_empire
# (copie exacte 4.4.6 + le type adastra_grounded, pour que districts, bâtiments,
# zones et le reste de l'interface traitent l'empire verrouillé comme un empire normal.
# Le filet de sécurité des avant-postes teste "is_country_type = default" en dur
# et continue donc de nous ignorer.)

is_regular_empire = {
	OR = {
		is_country_type = default
		#for checks in buildings and districts
		is_country_type = awakened_marauders
		is_country_type = fallen_empire
		is_country_type = awakened_fallen_empire
		is_country_type = exiled
		is_country_type = adastra_grounded
	}
	NOT = { has_ethic = ethic_gestalt_consciousness	}
}

# 1.2 - qui paie l'entretien de ses districts en energie.
#
# Vrai pour TOUS les empires de la galaxie, sauf un empire Ad Astra qui n'a pas
# encore l'electricite : celui-la paie en minerais (voir la surcharge des
# districts). Ecrit de facon a ce que le cas normal soit la premiere condition,
# donc evalue en premier.
adastra_pays_energy_upkeep = {
	OR = {
		NOT = { has_origin = origin_adastra }
		adastra_has_energy = yes
	}
}

# 1.2 - surcharge du declencheur scripte vanilla is_low_tech_start
# (copie exacte 4.4.6 + notre origine pendant le confinement).
#
# TROUVE LE 15/08 EN AUDITANT LES TECHNOS ANACHRONIQUES. Le jeu de base a deja
# la notion exacte dont ce mod a besoin : « cet empire commence sans technique ».
# Elle couvre les primitifs, Chaines brisees, Revanche et les Explorateurs
# enthousiastes. Ad Astra n'y figurait pas, alors que c'est le cas d'ecole.
#
# Ce que la surcharge repare, verifie ligne par ligne dans les fichiers du jeu :
#
#   - events/game_start.txt (lignes 1154, 1307, 1534) : le vaisseau scientifique,
#     le vaisseau de construction et la flotte de depart ne sont crees que si
#     is_low_tech_start = no. Ils n'existeront donc plus DU TOUT, au lieu d'etre
#     crees puis effaces par adastra.8. error.log en portait la trace a chaque
#     partie - trois « invalid design for create_ship effect » et un
#     « Failed to create starbase », consequences de designs que nous avions
#     retires.
#
#   - common/technology/00_biogenesis_tech.txt : Mauleurs, Tisseurs et leurs
#     deux ameliorations portent « starting_potential = { is_low_tech_start = no
#     # So low tech starts don't start with it } ». Un empire Ad Astra a
#     vaisseaux vivants commencait donc l'Age de pierre avec la technologie des
#     Mauleurs. Le commentaire de Paradox dit mot pour mot ce qu'on veut.
#
#   - common/pop_jobs/09_first_contact_jobs.txt : les emplois rudimentaires
#     (bureaucrate, pretre, artisan primitifs) exigent is_low_tech_start = yes.
#     Ils deviennent disponibles - ils restent conditionnes a leur batiment.
#
# La condition tombe A L'EMERGENCE : un empire qui a perce n'est plus un depart
# sans technique, et rien ne doit lui rester de ce statut.
is_low_tech_start = {
	OR = {
		is_primitive = yes
		has_origin = origin_broken_shackles
		has_origin = origin_payback
		is_eager_explorer_empire = yes
		AND = {
			has_origin = origin_adastra
			NOT = { has_country_flag = adastra_completed }
		}
	}
}

# 1.3 - Les technologies fondatrices du jeu de base.
#
# A l'Age spatial, elles ne sont plus offertes : elles se cherchent, au meme
# titre que les technologies d'epoque. Le Programme d'hyperpropulsion ne
# s'ouvre que lorsque les seize sont trouvees - c'est ce qui donne une raison
# de toutes les prendre au lieu de courir droit vers la sortie.
adastra_fondatrices_toutes = {
	has_technology = tech_space_exploration
	has_technology = tech_thrusters_1
	has_technology = tech_space_construction
	has_technology = tech_corvettes
	has_technology = tech_mass_drivers_1
	has_technology = tech_ship_armor_1
	has_technology = tech_shields_1
	has_technology = tech_starbase_1
	has_technology = tech_starbase_2
	has_technology = tech_space_defense_station_1
	has_technology = tech_solar_panel_network
	has_technology = tech_colonization_1
	has_technology = tech_interplanetary_commerce
	has_technology = tech_hydroponics
	has_technology = tech_holo_entertainment
	has_technology = tech_reactor_boosters_1
}

# ============================================================================
# 1.3 - LE JEU AVANT LA TECHNOLOGIE
#
# Demande d'ampynjord, 16/08 : a l'Age spatial, une technologie du jeu de base
# ne doit etre cherchable que si ce qu'elle sert est effectivement jouable.
# Chercher l'amelioration des corvettes quand on ne peut construire aucun
# vaisseau, c'est de la recherche morte - le joueur depense son budget dans une
# carte qui ne changera rien avant des dizaines d'annees.
#
# Chaque groupe de fondatrices porte donc la condition du jeu qu'il ouvre.

# Ce qui ouvre le Programme d'exploration : la phase d'exploration de la
# situation. Retour de test du 16/08 : « celle-la doit spawn quand j'ai atteint
# la phase d'exploration spatiale, c'est la tech necessaire pour debloquer la
# decision ». Avant cette phase, l'Age spatial a ses vingt-cinq technologies
# d'epoque - il y a de quoi chercher.
adastra_gameplay_orbite = {
	has_country_flag = adastra_unlock_space
	has_country_flag = adastra_stage_explore
}

# Le vaisseau de construction et les stations : c'est la phase de la base
# stellaire qui les ouvre, puisque c'est sa decision qu'ils servent.
adastra_gameplay_chantier = {
	has_country_flag = adastra_unlock_space
	has_country_flag = adastra_stage_orbital
}

# Les vaisseaux : il faut pouvoir en faire. Deux chemins - le chantier orbital
# une fois la base stellaire batie, ou le Programme de flotte militaire, qui
# livre des corvettes sans chantier. L'etape orbitale ouvre les deux.
adastra_gameplay_vaisseaux = {
	has_country_flag = adastra_unlock_space
	OR = {
		has_country_flag = adastra_stage_orbital
		has_country_flag = adastra_phase2_done
	}
}

# La base stellaire et ses modules : il faut l'avoir batie, pas seulement
# pouvoir la commander. Ameliorer un port stellaire qu'on n'a pas n'a pas de
# sens, et l'evenement « Le ciel est deja pris » rend ce cas tout a fait reel.
adastra_gameplay_base = {
	has_country_flag = adastra_unlock_space
	has_country_flag = adastra_phase2_done
}

# Coloniser : le Programme de colonisation s'ouvre a l'etape orbitale.
adastra_gameplay_colonie = {
	has_country_flag = adastra_unlock_space
	has_country_flag = adastra_stage_orbital
}

# Ce qui se joue entierement au sol. Ouvert a la phase d'exploration et non a
# l'entree dans l'Age spatial : l'age a deja ses vingt-cinq technologies
# d'epoque pour occuper le joueur, alors que la phase d'exploration n'en avait
# que deux. Retour de test du 16/08 : « faudrait pas que j'aie rien pendant
# longtemps, c'est frustrant ».
adastra_gameplay_sol = {
	has_country_flag = adastra_unlock_space
	has_country_flag = adastra_stage_explore
}
