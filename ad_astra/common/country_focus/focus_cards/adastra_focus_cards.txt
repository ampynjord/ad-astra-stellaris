# Ad Astra - cartes d'objectifs dédiées à l'ascension pré-PRL

# Atteindre l'âge de la vapeur
adastra_focus_100 = {
	category = development
	tier = 1
	background = "GFX_focus_card_background_development_tier_1"

	name = "adastra_focus_100.name"
	desc = "adastra_focus_100.desc"
	hint = "adastra_focus_100.hint"
	lore = "adastra_focus_100.lore"

	possible = {
		has_origin = origin_adastra
		has_country_flag = adastra_locked
	}

	condition_for_achievement = {
		custom_tooltip = {
			fail_text = "adastra_focus_100.fail"
			success_text = "adastra_focus_100.success"
			has_country_flag = adastra_reached_steam
		}
	}

	on_success = {
		create_focus_completed_message = { CATEGORY = DEVELOPMENT }
		add_focus_progress = { category = development amount = 10 }
	}
}
