### zone_industrial -> tech_adastra_steam_engine
zone_industrial = {
	icon = GFX_district_specialization_industrial
	base_buildtime = @zone_buildtime
	potential = { # portee planete
		# Ad Astra : Industrie mixte : la machine a vapeur permet la premiere industrie moderne
		OR = {
			NOT = { exists = owner }

				OR = {
					NOT = { has_origin = origin_adastra }
