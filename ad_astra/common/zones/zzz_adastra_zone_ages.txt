### zone_industrial -> steam
zone_industrial = {
	icon = GFX_district_specialization_industrial
	base_buildtime = @zone_buildtime
	potential = { # portee planete
		# Ad Astra : Industrie mixte : la Machine a vapeur, premiere vraie industrie
		OR = {
			NOT = { exists = owner }
			owner = {
				OR = {
					NOT = { has_origin = origin_adastra }
					has_country_flag = adastra_completed
					has_country_flag = adastra_reached_steam
				}
			}
		}
 # planet scope
		exists = owner
		owner = {
			country_uses_consumer_goods = yes
		}
	}

	unlock = { # portee planete
		# Ad Astra : Industrie mixte : la Machine a vapeur, premiere vraie industrie
		OR = {
			NOT = { exists = owner }
			owner = {
				OR = {
					NOT = { has_origin = origin_adastra }
					has_country_flag = adastra_completed
					has_country_flag = adastra_reached_steam
				}
			}
		}
 # planet scope
		hidden_trigger = {
			exists = owner
			owner = {
				country_uses_consumer_goods = yes
			}
		}
	}

	resources = {
		category = planet_zones
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = alloys
			COST = @zone_cost
			NOMADIC_COST_MULT = 0.5
		}
	}

	included_building_sets = {
		industrial
		urban_automation
		origin
	}

	inline_script = {
		script = jobs/zone_foundry_add
		AMOUNT = @scaling_district_2_jobs
		AMOUNT2 = @scaling_district_2_jobs
	}

	inline_script = {
		script = jobs/zone_factory_add
		AMOUNT = @scaling_district_2_jobs
		AMOUNT2 = @scaling_district_2_jobs

		urban

	}
