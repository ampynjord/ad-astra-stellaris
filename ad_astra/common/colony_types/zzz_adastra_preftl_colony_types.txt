# Ad Astra 1.4 (integration pre-PRL) - 2026-08-28
#
# Surcharge des 10 designations de colonie pre-PRL vanilla
# (`common/colony_types/00_colony_types.txt`, l. 2955-3103).
#
# Le vanilla les reserve a `is_country_type = primitive`. On elargit au type
# du joueur confine, sans rien changer d'autre : meme icone, meme poids.
# Elles sont purement cosmetiques cote vanilla (aucun modificateur, aucun
# emploi) mais elles donnent au monde natal l'ecusson de son epoque, ce qui
# est exactement ce qu'on veut voir dans l'interface de la planete.
#
# La condition `has_pre_ftl_age` est satisfaite par la couche A
# (`adastra.160`). Sans elle, aucune de ces designations n'apparaitrait.

@adastra_preftl_poids = 1000

col_pre_ftl_stone = {
	icon = GFX_colony_type_pre_ftl_stone
	potential = {
		owner? = {
			OR = { is_country_type = primitive  is_country_type = adastra_grounded }
			has_pre_ftl_age = stone_age
		}
	}
	weight_modifier = { base = @adastra_preftl_poids }
}
