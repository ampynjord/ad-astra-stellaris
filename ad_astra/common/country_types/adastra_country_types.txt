# Type du JOUEUR pendant le verrouillage : identique a default (jouable, tous modules)
# mais hors de portee du filet de securite qui reconstruit les avant-postes,
# et a l'abri des declarations de guerre.

adastra_grounded = {
	# 1.3 - LES VAISSEAUX NE SONT PLUS EXPULSES DE LEUR PROPRE CIEL.
	#
	# Quand un empire revendique le systeme natal, nos vaisseaux s'y retrouvent
	# en territoire etranger : le moteur les traite en intrus, leur donne l'ordre
	# de partir et les teleporte hors du systeme au bout du compte a rebours.
	# Un empire confine n'a nulle part ou aller - ses vaisseaux disparaissent
	# donc, et le programme spatial avec eux.
	#
	# `needs_border_access = no` est le mot-cle documente du jeu de base
	# (common/country_types/00_country_types.txt, ligne 49) : « decides if a
	# country needs to have explicit border access from another country to enter
	# its borders ». Sans lui, on n'a aucun moyen de garder ses propres vaisseaux
	# au-dessus de sa propre planete.
	# Ad Astra 1.4 (integration pre-PRL) 2026-08-28 - LE SEUL CHANGEMENT DE FOND DE CETTE BUILD.
	#
	# Le moteur ne reconnait pas un pre-PRL a son country_type mais au booleen
	# `faction.primitive`, que lit le trigger natif `is_primitive` (506 usages
	# vanilla, defini dans aucun fichier de script : il est en dur). La doc de
	# Paradox (00_country_types.txt l. 44) precise que les valeurs du bloc
	# `faction` « sont copiees sur le pays et modifiables en cours de partie ».
	#
	# Rien dans le vanilla ne lie ce champ a `playable` : ce sont deux champs
	# independants. C'est l'hypothese que cette build sert a verifier, et rien
	# d'autre - tout le reste du type est laisse strictement identique pour que
	# le resultat soit attribuable a ce seul champ.
	#
	# On ne pose PAS `primitive_age` : le type primitive vanilla ne le pose pas
	# non plus, il n'est que documente en commentaire. Champ non verifie, retire.
	faction = {
		needs_border_access = no
		generate_borders = yes
		needs_colony = yes
		auto_delete = no
		primitive = yes
		# 1.4 (2026-08-28 - deuxieme champ repris du bloc `faction` du type
		# `primitive` vanilla (00_country_types.txt l. 857-864).
		#
		# Il gouverne l'application des effets d'intel a la prospection. Sans
		# lui, l'empire est primitif pour le moteur mais garde le comportement
		# d'intel d'un empire normal - ce qui expliquait les astres qui se
		# prospectaient seuls, les etoiles decouvertes deja prospectees et le
		# vaisseau scientifique qui semblait prospecter instantanement.
		#
		# Le vanilla pose aussi `generate_borders = no`, NON repris ici : il
		# toucherait au correctif 1.3 des vaisseaux expulses de leur propre
		# ciel. A discuter separement.
		intel_effects_surveyed = no
	}
	has_difficulty_bonuses = yes
	counts_for_victory = no
	pop_growth = yes
	playable = yes
	has_pulse_events = yes
	has_deficits = yes
	piracy_risk = yes
	randomize_ethos = yes
	shuffle_pop_ethos = yes
	needs_survey = yes
	# 1.2 : la guerre redevient possible dans les deux sens. Une civilisation
	# pre-PRL peut declarer la guerre a un empire stellaire - et surtout, elle
	# peut en subir une. C'est le risque du chemin « galaxie vivante », celui
	# que l'Heritage de l'Ascension recompense desormais au double.
	diplomatic_wars = yes
	ship_disengagement = yes
	army_disengagement = yes
	relations = yes
	# Le pays confine ne transmet pas son reseau de contacts a la galaxie.
	# Ses rencontres restent cependant des premiers contacts vanilla : elles
	# doivent pouvoir creer leurs indices, leurs evenements et leur chaine.
	share_communications = no
	# 31/08 - une civilisation au sol ne figure dans le carnet d'adresses de
	# personne : un tiers ne peut pas nous « presenter », et nos releves du
	# ciel restent les notres.
	can_have_their_comms_shared = no
	share_survey_data = no
	observable = yes
	can_use_invalid_technology = no
	can_receive_envoys = yes
	ownership_limits = default
	do_critters_age = yes

	ai = {
		enabled = yes
		declare_war = yes
		min_navy_for_wars = 0.5
		min_assault_armies_for_wars = @armies_for_declaring_war
		colonizer = yes
		internal_policies = yes
		modification = yes
		government = yes
		traditions = yes
		ascension_perks = yes
		construction = yes
		follow = yes
		can_use_market = yes
		modules = {
			military_minister_module
			foreign_minister_module
			interior_minister_module
			mercenary_buyer_minister_module
			enclave_founder_minister_module
		}
	}

	trade_routes_available = {
		is_gestalt = no
	}

	modules = {
		standard_event_module = {}
		standard_economy_module = {}
		standard_leader_module = {}
		# Meme regle que le type default : les rencontres avant l'emergence
		# doivent creer les sites de premier contact vanilla, sans partage
		# automatique de la carte ni du reseau diplomatique.
		standard_diplomacy_module = { contact_rule = does_first_contact_sites }
		standard_technology_module = {}
		standard_pop_factions_module = {}
		standard_expansion_module = {}
		standard_species_rights_module = {}
		standard_grand_archive_module = {}
		standard_shroud_module = {}
	}

	resources = {
		category = country_base
		# 1.2 : une ressource n'existe pas avant la techno qui l'invente.
		# A l'Age de pierre on a des minerais, de la nourriture, du commerce,
		# de l'influence, de l'unite et de la recherche - rien d'autre.
		# Les alliages arrivent avec la Metallurgie du bronze, les biens de
		# consommation avec la Machine a vapeur (la Manufacture est le premier
		# atelier qui produit en serie), l'energie avec le Reseau electrique.
		produces = {
			physics_research = 10
			society_research = 10
			engineering_research = 10
			influence = 3
			unity = 10
		}
		produces = {
			trigger = { adastra_has_alloys = yes }
			alloys = 5
		}
		produces = {
			trigger = { adastra_has_energy = yes }
			energy = 20
		}
		produces = {
			trigger = {
				is_nomadic = no
			}
			minerals = 20
		}
		produces = {
			trigger = {
				adastra_has_consumer_goods = yes
				OR = {
					is_gestalt = no
					has_valid_civic = civic_machine_servitor
				}
			}
			consumer_goods = 15
		}

		# 1.2 - PANSEMENT RETIRE le 15/08.
		#
		# Il produisait 2 biens de consommation par mois tant que la Machine a
		# vapeur manquait, pour couvrir les 10 % d'entretien que le plafond du
		# moteur (-90 %) refusait d'annuler. En jeu, il sur-corrigeait : la barre
		# affichait « 89 (+1) » a l'age de pierre, soit une ressource qui n'est
		# pas censee exister, en train de s'accumuler sous les yeux du joueur.
		#
		# « Chacun fabrique le sien » empile desormais le multiplicateur global
		# des emplois ET celui de chaque categorie d'emploi. Si les plafonds
		# s'appliquent multiplicateur par multiplicateur, le residu tombe a 1 %
		# et la compensation n'a plus lieu d'etre.
		#
		# A VERIFIER EN JEU : si les biens de consommation passent au rouge a
		# l'age de pierre, c'est que le moteur additionne les multiplicateurs au
		# lieu de les composer. Dans ce cas, retablir ce bloc a 1 (pas 2).
		produces = {
			trigger = {
				OR = { #Either
					# You're a non-lithoid non-individual machine, regular empire or hive mind
					AND = {
						is_individual_machine = no
						is_lithoid_empire = no
						is_infernal_empire = no
						OR = {
							is_regular_empire = yes
							is_hive_empire = yes
						}
					}
					# You are a lithoid empire with a non-lithoid subspecies
					AND = {
						is_lithoid_empire = yes
						has_country_flag = non_lithoid_subspecies
					}
					# Or a machine empire with a non-lithoid secondary species
					AND = {
						has_country_flag = non_lithoid_subspecies
						is_machine_empire = yes
					}
					# Or an individual machine with a non-lithoid species
					AND = {
						is_individual_machine = yes
						has_country_flag = non_lithoid_subspecies
					}
					# Or have bioships
					country_uses_bio_ships = yes
				}
			}
			food = 20
		}
		produces = {
			trigger = {
				# Either
				OR = {
					# You're a lithoid regular empire or hive mind
					AND = {
						is_lithoid_empire = yes
						OR = {
							is_regular_empire = yes
							is_hive_empire = yes
						}
					}
					# Or a machine empire with a lithoid secondary species
					AND = {
						has_country_flag = lithoid_subspecies
						is_machine_empire = yes
					}
					# Or a machine empire with no secondary species
					AND = {
						is_machine_empire = yes
						NOR = {
							has_country_flag = lithoid_subspecies
							has_country_flag = non_lithoid_subspecies
						}
					}
				}
			}
			minerals = 20
		}
		produces = {
			trigger = {
				is_machine_empire = yes
				adastra_has_energy = yes
			}
			energy = 20
		}
		produces = {
			trigger = {
				is_lithoid_empire = yes
				has_valid_civic = civic_hive_cordyceptic_drones
			}
			food = 10
			minerals = -10
		}
		produces = {
			trigger = {
				adastra_has_alloys = yes
				OR = {
					is_infernal_empire = yes
					# Or a machine with an infernal species
					AND = {
						has_country_flag = infernal_subspecies
						is_machine_empire = yes
					}
				}
			}
			alloys = 10
		}
	}

	# planet scope

		category = planet_branch_offices

			trade = 1000
		}
