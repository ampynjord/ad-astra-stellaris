# Ad Astra 1.4 - la sonde profonde.
#
# Une civilisation confinee ne peut pas EXPLORER, mais elle peut REGARDER, et
# elle peut ENVOYER. Chaque sonde reconnait UN astre encore inconnu du systeme
# natal, et en revele les gisements.
#
# ESSAI 2026-08-29 - TROIS CHANGEMENTS DEMANDES PAR AMPYNJORD :
#
# 1. REPETABLE. La sonde n'est plus unique : tant qu'il reste un astre inconnu
#    dans le systeme natal, on peut en lancer une autre. Le drapeau
#    `core_sonde_profonde_fait` reste pose (la chaine des jalons s'appuie
#    dessus) mais ne ferme plus la decision.
# 2. COUT PROGRESSIF. Chaque lancement rencherit le suivant de 35 %, sur le
#    modele des relances du programme d'exploration
#    (`adastra_cout_relance_explore`). Une campagne de reconnaissance complete
#    du systeme doit couter, pas se cliquer.
# 3. LE JALON LUNAIRE N'EST PLUS UN CUL-DE-SAC. Le premier atterrissage exige
#    une lune dans le systeme natal ; sur un systeme procedural sans lune, la
#    decision n'apparait jamais et bloquait definitivement la sonde, donc toute
#    la progression. Il reste OBLIGATOIRE quand une lune existe, et devient
#    sans objet quand il n'y en a pas.
#
# Le test de lune se fait depuis la RACINE (garantie capitale par
# `is_capital = yes`), jamais par `capital_scope` : chez un empire nomade ce
# dernier renvoie un vaisseau et tout `solar_system` qui suit echoue en boucle.

decision_core_sonde_profonde = {
	owned_planets_only = yes
	icon = decision_core_sonde_profonde

	# Trois ans de conception, d'integration et de trajectoire.
	enactment_time = 360

	resources = {
		category = decisions
		cost = {
			influence = 300
			minerals = 1000
			alloys = 100
			mult = value:adastra_cout_relance_sonde
		}
	}

	potential = {
		is_capital = yes
		owner = {
			core_est_confine = yes
			core_poussee_3 = yes
			has_country_flag = core_vol_habite_fait
		}
		# Le jalon lunaire, obligatoire seulement s'il est possible.
		OR = {
			owner = { has_country_flag = core_premier_corps_fait }
			NOT = { solar_system = { any_system_planet = { is_moon = yes } } }
		}
		# Il reste quelque chose a reconnaitre.
		solar_system = {
			any_system_planet = {
				is_star = no
				NOT = { is_same_value = root }
				NOT = { is_surveyed = { who = root.owner status = yes } }
			}
		}
	}

	effect = {
		custom_tooltip = decision_core_sonde_tooltip
		hidden_effect = {
			owner = { country_event = { id = adastra.110 } }
		}
	}

	ai_weight = {
		weight = 40
	}
}
