# Ad Astra - le programme spatial en trois phases
# Chaque décision correspond à une étape de la situation Au-delà des étoiles.

# Phase 1 : programme d'exploration spatiale (vaisseau scientifique sous-luminique)
decision_adastra_explore = {
	owned_planets_only = yes
	icon = decision_adastra_explore

	enactment_time = 360

	# 1.3 : le prix. Les programmes coutaient de l'energie, c'est-a-dire une
	# ressource qui n'existe pas avant l'electricite et qu'on a donc en tas a
	# l'Age spatial. Ils sont desormais payes en INFLUENCE et en MINERAIS :
	# monter un programme spatial est d'abord une decision politique, et
	# l'influence est la seule ressource vraiment rare d'un empire au sol.
	resources = {
		category = decisions
		cost = {
			influence = 250
			minerals = 800
			mult = value:adastra_cout_relance_explore
		}
	}

	# REPETABLE, depuis le 16/08. La decision se fermait apres son premier
	# succes : un vaisseau scientifique perdu - accident du Voile, empire
	# hostile, mauvaise manoeuvre - l'etait pour toute la partie tant que le
	# chantier orbital n'existait pas, c'est-a-dire pendant toute la phase 1.
	# C'etait le cul-de-sac signale par Sithiya sur le Workshop. On peut
	# desormais relancer le programme autant de fois qu'on en paie le prix ;
	# la porte se ferme a l'emergence, quand les vaisseaux du jeu de base
	# prennent le relais.
	potential = {
		is_capital = yes
		owner = {
			has_country_flag = adastra_stage_explore
			# 1.4 : l'echelle de propulsion. Un vaisseau qui doit atteindre
			# les autres astres du systeme demande une poussee qui dure -
			# l'ionique de 1964. La fondatrice reste exigee, elle n'est
			# simplement plus seule a decider.
			core_poussee_3 = yes
			has_technology = tech_space_exploration
			NOT = { has_country_flag = adastra_completed }
		}
	}

	effect = {
		# 1.3 : surcout cumulatif. Un programme relance coute plus cher que
		# le precedent - monter une seconde expedition n'a jamais ete aussi
		# simple que la premiere, et ca empeche d'en enchainer dix.
		hidden_effect = { owner = { change_variable = { which = adastra_relances_explore value = 1 } } }
		custom_tooltip = decision_adastra_explore_tooltip
		hidden_effect = {
			owner = {
				country_event = { id = adastra.85 }
			}
		}
	}

	ai_weight = {
		weight = 100
	}
}

# La base stellaire reste un acte du joueur dans la vue systeme : le programme
# ne doit jamais la creer gratuitement ni contourner son cout vanilla.

	owned_planets_only = yes
	icon = decision_adastra_starbase

	enactment_time = 720

			minerals = 2000
			alloys = 400
		}
	}

		owner = {
			has_country_flag = adastra_stage_constructor

			# d'eclat - ionique, bouclier ablatif, supraconductivite.
			core_poussee_4 = yes
			# 31/08 - retour ampynjord (save 2244.04.29) : Batisseur detruit,

# Facultatif : programme de flotte militaire (disparaît à la découverte de l'Hyperpropulsion)
decision_adastra_navy = {
	owned_planets_only = yes
	icon = decision_adastra_navy

	enactment_time = 240

	resources = {
		category = decisions
		# 1.2 : un empire a bio-vaisseaux ne taille pas ses corvettes dans la
		# roche, il les fait pousser. Le cout en minerais devient un cout en
		# nourriture (retour Sithiya : « des vaisseaux vivants faits de cailloux »).
		cost = {
			trigger = {
				owner = { country_uses_bio_ships = no }
			}
			energy = 400
			minerals = 200
			alloys = 300
		}
		cost = {
			trigger = {
				owner = { country_uses_bio_ships = yes }
			}
			energy = 400
			food = 200
			alloys = 300
			mult = value:adastra_cout_relance_navy
		}
	}

	# 16/08 : la garde exigeait adastra_phase2_done, c'est-a-dire que le
	# CHANTIER ORBITAL soit bati. Un empire dont le ciel est occupe ne peut pas
	# le batir - et perdait du meme coup la flotte militaire et la colonisation.
	# Un programme ne doit dependre que de l'ETAPE atteinte, jamais de la
	# reussite d'un autre programme : les vaisseaux sont livres par evenement,
	# ils n'ont jamais eu besoin d'un chantier pour naitre.
	potential = {
		is_capital = yes
		owner = {
			# 1.4 : meme poussee que le chantier. Une coque de guerre n'est
			# pas plus facile a mettre en orbite qu'une coque civile.
			core_poussee_4 = yes
			has_technology = tech_corvettes
			NOT = { has_technology = tech_hyper_drive_1 }
		}
	}

	effect = {
		# 1.3 : surcout cumulatif. Un programme relance coute plus cher que
		# le precedent - monter une seconde expedition n'a jamais ete aussi
		# simple que la premiere, et ca empeche d'en enchainer dix.
		hidden_effect = { owner = { change_variable = { which = adastra_relances_navy value = 1 } } }
		custom_tooltip = decision_adastra_navy_tooltip
		hidden_effect = {
			owner = {
				country_event = { id = adastra.87 }
			}
		}
	}

	# 1.2 : facultatif pour le joueur, indispensable pour l'IA. Depuis que la
	# guerre est possible dans les deux sens, un empire IA confine sans la
	# moindre corvette est une proie gratuite. Elle le prend des qu'elle en a
	# les moyens, avec priorite absolue si elle est deja en guerre.
	ai_weight = {
		weight = 60
		modifier = {
			factor = 3
			owner = { is_at_war = yes }
		}
		modifier = {
			factor = 0
			owner = { has_monthly_income = { resource = alloys value < 1 } }
		}
	}
}

# 1.1 : Campagnes d'exploitation - collecter plus, au prix d'un compromis.
# Chaque levier arrive a l'epoque qui l'a vu naitre. Modificateurs temporaires (10 ans), non cumulables.

decision_adastra_campaign_harvest = {
	owned_planets_only = yes
	sound = event_administrative_work
	icon = decision_adastra_campaign_harvest

	enactment_time = 60

	# 1.2 : le prix suit l'age - 5 avant le Medieval, 10 ensuite, 25 a partir de
	# l'Industriel, 38 a l'Age spatial. Les 20 unite fixes de la 1.1
	# representaient 17 mois de revenu a l'Age de pierre : la campagne etait
	# inaccessible la ou elle aurait ete la plus utile.
	#
	# 1.2 (correctif 13/08) : ecrit d'abord avec des blocs « modifier » dans
	# resources, que le moteur refuse - « Unexpected token: modifier » a chaque
	# chargement, et le prix restait bloque a 5 pour toute la partie. La seule
	# forme acceptee est une suite de blocs cost mutuellement exclusifs, comme
	# le fait deja le programme de flotte pour les bio-vaisseaux.
	resources = {
		category = decisions
		cost = {
			influence = 150
			minerals = 600
			alloys = 200
		}
	}

	potential = {
		# Systeme remplace en 1.4 : conserve seulement pour que les anciennes
		# sauvegardes puissent expirer leur modificateur sans erreur.
		always = no
		is_capital = yes
		owner = { has_country_flag = adastra_locked }
	}

	allow = {
		# 1.2 : une seule campagne a la fois - c'est un arbitrage,
		# pas un cumul de bonus.
		NOT = { has_modifier = adastra_campaign_mining }
		NOT = { has_modifier = adastra_campaign_fuel }
		NOT = { has_modifier = adastra_campaign_industry }
		NOT = { has_modifier = adastra_campaign_harvest }
	}

	effect = {
		add_modifier = { modifier = adastra_campaign_harvest days = 3600 }
	}

	# 1.2 : l'IA peut prendre l'origine, elle doit donc savoir arbitrer entre
	# les quatre campagnes. Chacune est un echange : +40%% sur une ressource,
	# -25%% sur une autre. On ne l'active donc que si la ressource visee est
	# reellement en tension, et jamais si celle qui paie l'est deja.
	ai_weight = {
		weight = 0
		modifier = {
			add = 50
			owner = { has_monthly_income = { resource = food value < 3 } }
		}
		modifier = {
			add = 50
			owner = { has_monthly_income = { resource = food value < 0 } }
		}
		modifier = {
			factor = 0
			owner = { has_monthly_income = { resource = minerals value < 2 } }
		}
	}
}

# Programme de colonisation (relancable) - ajoute le 16/08.
#
# L'Arche etait livree en PLAN par le programme de base stellaire, et rien de
# plus. Un plan de colonisateur ne se construit pas au chantier : dans le jeu
# de base, un vaisseau de colonisation nait du bouton « Coloniser » pose sur
# une cible. Un joueur qui batissait un habitat du vide au-dessus de son monde
# n'avait donc aucun moyen evident d'aller le peupler, et le plan restait une
# ligne morte dans sa liste de designs.
#
# On en fait un programme, comme les trois autres : il livre un vaisseau, de
# l'espece principale, pret a partir. Relancable, parce qu'un monde natal peut
# avoir plus d'une cible - une seconde planete habitable dans le systeme, ou
# autant d'habitats du vide qu'on sait en batir.
decision_adastra_colony = {
	owned_planets_only = yes
	icon = decision_adastra_colony

	enactment_time = 540

	resources = {
		category = decisions
		# Meme partage que la flotte militaire : un empire a vaisseaux vivants
		# ne taille pas sa coque dans la roche, il la fait pousser.
		cost = {
			trigger = {
				owner = { country_uses_bio_ships = no }
			}
			energy = 500
			minerals = 400
			alloys = 200
		}
		cost = {
			trigger = {
				owner = { country_uses_bio_ships = yes }
			}
			energy = 500
			food = 400
			alloys = 200
		}
	}

	# Meme raison que la flotte militaire : l'etape, pas le chantier.
	potential = {
		is_capital = yes
		# 31/08 - les decisions sont aussi evaluees sur les colonies-vaisseaux
		# (nomades) : sans cette garde, « Invalid context switch [solar_system]
		# from ship » a chaque evaluation (error.log, partie 14).
		exists = solar_system
		# ESSAI 2026-08-30 - le programme n'apparait que s'il a une cible :
		# un monde du systeme natal theoriquement colonisable et sans
		# proprietaire. Sans cible, la decision n'a pas de sens.
		solar_system = {
			any_system_planet = {
				is_colonizable = yes
				NOT = { exists = owner }
			}
		}
		owner = {
			has_country_flag = adastra_stage_orbital
			# 1.3 : comme les trois autres programmes, celui-ci exige sa
			# technologie fondatrice. Livrer une Arche a un empire qui ignore
			# le Protocole nouveaux mondes n'avait aucun sens.
			has_technology = tech_colonization_1
			NOT = { has_country_flag = adastra_completed }
		}
	}

	effect = {
		custom_tooltip = decision_adastra_colony_tooltip
		hidden_effect = {
			owner = {
				country_event = { id = adastra.88 }
			}
		}
	}

	# L'IA ne s'en sert que si elle a quelque chose a peupler.
	ai_weight = {
		weight = 0
	}
}
