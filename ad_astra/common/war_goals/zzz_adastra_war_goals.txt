# Ad Astra 1.3 - casus belli et buts de guerre elargis a l'empire au sol.
#
# L'audit du 16/08 : is_country_type = default est ecrit en dur 39 fois dans les
# casus belli du jeu de base et 29 fois dans les buts de guerre. Tant que la
# guerre etait impossible ca n'avait aucune consequence ; la 1.2 l'a rouverte
# dans les deux sens, et le probleme est devenu reel.
#
# Deux directions, verifiees bloc par bloc :
#   - la garde porte sur ROOT (l'attaquant) : cb_claim.potential, par exemple.
#     Un empire au sol ne peut donc revendiquer personne.
#   - la garde porte sur FROM (la cible) : wg_humiliation.possible, par exemple.
#     Un empire qui nous declare la guerre n'a alors AUCUN but applicable, et
#     on ne sait pas ce que le moteur en fait.
#
# On reprend le bloc vanilla complet et on elargit la condition, jamais on ne
# remplace : c'est la methode deja employee pour les actions diplomatiques, les
# designations de capitale et la Grande Archive.
#
# PERIMETRE ASSUME : les quinze entrees d'une guerre ordinaire entre un empire
# regulier et nous. Les casus belli lies a un DLC ou a une origine particuliere
# (colosse, Galatron, Scion, Hegemon, Imperium, nomades, corporations) sont
# laisses tels quels : ils ne sont pas atteignables depuis le sol, et les
# surcharger multiplierait la surface de conflit avec les autres mods pour rien.
#
# FICHIER GENERE PAR tools/gen_war_overrides.py - NE PAS EDITER A LA MAIN.

wg_plunder = {
	casus_belli = cb_despoliation
	total_war = no

	galactic_empire_joins_defender = {
		from = { is_galactic_emperor = yes }
		imperial_authority >= 100
	}

	surrender_acceptance = -25

	potential = {
		has_total_war_cb = no
		has_valid_civic = civic_barbaric_despoilers
		NOT = { is_in_federation_with = from }
		NAND = {
			is_galactic_emperor = yes
			from = { is_galactic_community_member = yes }
		}
	}
	possible = {
		custom_tooltip = {
			fail_text = has_claimed_all_systems
			from = {
				OR = {
					is_country_type = default
					# Ad Astra : un empire au sol reste un empire.
					is_country_type = adastra_grounded
				}
				any_owned_planet = {
					is_colony = yes
					root = {
						NOT = { has_claim = prev.solar_system }
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = cannot_use_against_emperor
			OR = {
				from = { is_galactic_emperor = no }
				is_galactic_community_member = no
			}
		}
		custom_tooltip = {
			fail_text = pax_galactica_active
			NAND = {
				is_active_resolution = resolution_emperor_pax_galactica
				is_galactic_community_member = yes
				from = { is_galactic_community_member = yes }
			}
		}
	}
	on_accept = {
		custom_tooltip = wg_plunder_accept_them
		add_resource = {
			energy = 250
			minerals = 250
			mult = from.trigger:num_owned_planets
		}
		hidden_effect = { from = { add_modifier = { modifier = raided days = 3600 } } }
	}

	ai_weight = {
		weight = 100
	}
}

	on_accept = {
		hidden_effect = {
			from = {
				add_threat = {
					who = root
					amount = 0.5
				}
			}
		}
		if = {
			limit = { from = { has_been_declared_crisis = no } }
			from = {
				if = {
					limit = {
						tech_unlocked_ratio = {
							who = ROOT
							ratio < @protectorate_tech_threshold
						}
					}
					set_subject_of = {
						who = ROOT
						preset = preset_protectorate
					}
				}
				else = {
					ROOT = {
						switch = {
							trigger = has_policy_flag
							oppressive_terms = {
								if = {
									limit = {
										ROOT = { is_ai = yes }
									}
									FROM = {
										set_subject_of = {
											who = ROOT
											preset = preset_vassal_oppressive_ai
										}
									}
								}

										set_subject_of = {
											who = ROOT
											preset = preset_vassal_oppressive
										}
									}
								}
							}
							balanced_terms = {
								FROM = {
									set_subject_of = {
										who = ROOT
										preset = preset_vassal_balanced
									}
								}
							}
							benevolent_terms = {
								FROM = {
									set_subject_of = {
										who = ROOT
										preset = preset_vassal_benevolent
									}
								}
							}
						}
					}
				}
				add_modifier = {
					modifier = newly_subjugated_loyalty
					years = 5
				}
			}
		}
	}

	on_status_quo = {
		if = {
			limit = {
				exists = last_created_country
				last_created_country = {
					has_country_flag = released_empire
				}
			}

			last_created_country = {
				set_origin_effect = { SOURCE_COUNTRY = from }
				if = {
					limit = {
						tech_unlocked_ratio = {
							who = ROOT
							ratio < @protectorate_tech_threshold
						}
					}
					set_subject_of = {
						who = ROOT
						preset = preset_protectorate
					}
				}
				else = {
					ROOT = {
						switch = {
							trigger = has_policy_flag
							oppressive_terms = {
								if = {
									limit = {
										ROOT = { is_ai = yes }
									}
									PREV = {
										set_subject_of = {
											who = ROOT
											preset = preset_vassal_oppressive_ai
										}
									}
								}

									PREV = {
										set_subject_of = {
											who = ROOT

										}
									}
								}
							}
							balanced_terms = {
								PREV = {
									set_subject_of = {
										who = ROOT
										preset = preset_vassal_balanced
									}
								}
							}
							benevolent_terms = {
								PREV = {
									set_subject_of = {
										who = ROOT
										preset = preset_vassal_benevolent
									}
								}
							}
						}
					}
				}
				add_modifier = {
					modifier = newly_subjugated_loyalty
					years = 5
				}
			}

			if = {
				limit = {
					root = { is_megacorp = yes }
					last_created_country = { is_megacorp = yes }
				}
				last_created_country = {
					set_merchant_government_effect = yes
				}
			}
		}
	}

	ai_weight = {
		weight = 10
		modifier = {
			factor = 0
			has_ai_personality_behaviour = wants_tribute
		}
	}
