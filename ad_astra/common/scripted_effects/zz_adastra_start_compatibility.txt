# Les batiments explicitement fournis par un civisme reviennent des que leur
# economie est capable d'assurer leur entretien. L'effet est appele sur la
# capitale et ne remplace jamais un batiment existant.
adastra_grant_civic_infrastructure = {
	if = {
		limit = {
			owner = { has_civic = civic_reanimated_armies }
			owner = { has_country_flag = adastra_reached_steam }
			owner = { adastra_has_energy = yes }
			owner = { adastra_has_consumer_goods = yes }
			free_building_slots > 0
			NOT = { has_building = building_dread_encampment }
		}
		add_building = building_dread_encampment
	}
}
