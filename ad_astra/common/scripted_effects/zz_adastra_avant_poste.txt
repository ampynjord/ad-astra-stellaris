# Ad Astra - avant-poste du systeme natal (ecrit a la main, non genere).
#
# 11/11 - DECISION D'AMPYNJORD, DEFAUT BLOQUANT B09 (test en jeu beta.12).
# Sans base, le systeme natal n'a pas de frontieres et le moteur n'offre aucun
# ordre « Construire une base » (vue systeme ni vue galaxie). On pose donc une
# base minimale, au rang avant-poste, qui appartient au joueur ; il l'ameliore
# lui-meme (adastra.75 exige port stellaire, modules et batiment).
#
# Le design d'avant-poste n'existe pour un pays que s'il connait tech_starbase_1
# (error.log : « ship size starbase_outpost that doesn't have a design »). L'effet
# donne donc la technologie (prerequis tech_space_construction garanti par le
# garde ci-dessous), puis cree la base ; rappele a la recherche (adastra.132) et
# chaque mois (adastra.160). Idempotent : une base existante suffit.
adastra_creer_avant_poste = {
	# 11/11 - decision d'ampynjord : AUTOMATIQUE. La recherche automatique ne
	# choisissait pas Starbase Construction (le joueur devait penser a la
	# selectionner a la main). A la livraison du Batisseur, le prerequis
	# tech_space_construction est toujours acquis (fondation de l'etape
	# exploration) : on donne donc la technologie, qui ouvre le design de base.
	if = {
		limit = {
			has_country_flag = adastra_constructor_ready
			has_technology = tech_space_construction
			NOT = { has_technology = tech_starbase_1 }
		}
		give_technology = { tech = tech_starbase_1 message = no }
		set_country_flag = adastra_techbase_octroyee
	}
	if = {
		limit = {
			has_country_flag = adastra_constructor_ready
			has_technology = tech_starbase_1
			capital_scope = { solar_system = { NOT = { exists = starbase } } }
		}
		# Le pays confine n'a aucun design de base : on force la generation des
		# designs automatiques (ceux des tailles enable_default_design) avant de creer.
		refresh_auto_generated_ship_designs = yes
		capital_scope = {
			solar_system = {
				create_starbase = {
					size = starbase_outpost
					owner = root
				}
			}
		}
	}
}
