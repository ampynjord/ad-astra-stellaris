# Ad Astra 1.3 - casus belli et buts de guerre elargis a l'empire au sol.
#
# L'audit du 16/08 : is_country_type = default est ecrit en dur 39 fois dans les
# casus belli du jeu de base et 29 fois dans les buts de guerre. Tant que la
# guerre etait impossible ca n'avait aucune consequence ; la 1.2 l'a rouverte
# dans les deux sens, et le probleme est devenu reel.
#
# Deux directions, verifiees bloc par bloc :
#   - la garde porte sur ROOT (l'attaquant) : cb_claim.potential, par exemple.
#     Un empire au sol ne peut donc revendiquer personne.
#   - la garde porte sur FROM (la cible) : wg_humiliation.possible, par exemple.
#     Un empire qui nous declare la guerre n'a alors AUCUN but applicable, et
#     on ne sait pas ce que le moteur en fait.
#
# On reprend le bloc vanilla complet et on elargit la condition, jamais on ne
# remplace : c'est la methode deja employee pour les actions diplomatiques, les
# designations de capitale et la Grande Archive.
#
# PERIMETRE ASSUME : les quinze entrees d'une guerre ordinaire entre un empire
# regulier et nous. Les casus belli lies a un DLC ou a une origine particuliere
# (colosse, Galatron, Scion, Hegemon, Imperium, nomades, corporations) sont
# laisses tels quels : ils ne sont pas atteignables depuis le sol, et les
# surcharger multiplierait la surface de conflit avec les autres mods pour rien.
#
# FICHIER GENERE PAR tools/gen_war_overrides.py - NE PAS EDITER A LA MAIN.

cb_claim = {
	show_in_diplomacy = yes
	show_notification = no
	potential = {
		OR = {
			is_country_type = default
			# Ad Astra : un empire au sol reste un empire.
			is_country_type = adastra_grounded
		}
		NOT = { is_in_federation_with = from }
		is_nomadic = no
	}
	is_valid = {
		OR = {
			has_claim = FROM
			from = {
				any_subject = {
					root = { has_claim = prev }
				}
			}
		}
		has_total_war_cb = no
		NOT = { is_overlord_to = FROM }
		OR = {
			is_subject = no
			overlord = { NOT = { is_same_value = from } }
		}
	}
	proxy_war_resources = {
		category = proxy_war
		cost = {
			influence = 120
		}
	}
	on_proxy_war_start = {
		pay_proxy_war_additional_resources = { VALUE = -30 }
	}
}

cb_subjugation = {
	show_in_diplomacy = no
	show_notification = no
	potential = {
		OR = {
			OR = {
				is_country_type = default
				# Ad Astra : un empire au sol reste un empire.
				is_country_type = adastra_grounded
			}
			is_country_type = awakened_fallen_empire
		}
		NOT = { is_in_federation_with = from }
		is_subject = no
	}
	is_valid = {
		NOR = {
			is_overlord_to = FROM
			has_crisis_level = crisis_level_1
		}
		from = { is_subject = no }
		OR = {
			is_country_type = awakened_fallen_empire
			relative_power = {
				who = FROM
				category = all
				value > equivalent
			}
			AND = {
				is_nomadic = yes
				relative_power = {
					who = FROM
					category = fleet
					value > equivalent
				}
			}
		}
	}
	destroy_if = {
		NOT = { is_country_type = awakened_fallen_empire }
		has_crisis_level = crisis_level_1
		relative_power = {
			who = FROM
			category = all
			value < superior
		}
	}
	proxy_war_resources = {
		category = proxy_war
		cost = {
			influence = 200
		}
	}
	on_proxy_war_start = {
		pay_proxy_war_additional_resources = { VALUE = -50 }
	}
}

cb_humiliation = {
	show_in_diplomacy = yes
	show_notification = no
	potential = {
		OR = {
			OR = {
				is_country_type = default
				# Ad Astra : un empire au sol reste un empire.
				is_country_type = adastra_grounded
			}
			is_country_type = fallen_empire
		}
		NOT = { is_in_federation_with = from }
	}
	is_valid = {
		if = {
			limit = {
				NOT = {
					from = {
						has_relation_flag = {
							who = root
							flag = vivisectors
						}
					}
				}
			}
			OR = {
				NOT = { has_policy_flag = liberation_wars }
				from = { has_ethic = ethic_gestalt_consciousness }
			}
			OR = {
				is_rival = FROM
				AND = {
					OR = {
						is_active_resolution = "resolution_mutualdefense_castigation_proclamation"
						is_active_resolution = "resolution_mutualdefense_renegade_containment"
					}
					FROM = {
						OR = {
							has_modifier = resolution_denounced
							is_in_breach_of_any = yes
						}
					}
				}
			}
		}

		has_total_war_cb = no
		NOT = { is_overlord_to = FROM }
		OR = {
			is_subject = no
			overlord = { NOT = { is_same_value = from } }
		}
	}
	proxy_war_resources = {
		category = proxy_war
		cost = {
			influence = 100
		}
	}
	on_proxy_war_start = {
		pay_proxy_war_additional_resources = { VALUE = -25 }
	}
}
