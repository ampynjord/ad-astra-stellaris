# Ad Astra 1.2 - batiments d'epoque.
# FICHIER GENERE PAR tools/gen_age_buildings.py - NE PAS EDITER A LA MAIN.
# Source de verite : tools/age_buildings_data.py
#
# potential ne teste que l'origine : un batiment dont le potential devient
# faux est desactive par le moteur. Les batiments deja eleves doivent
# survivre a l'emergence, c'est donc allow qui ferme la construction.


# --- Monument des ancêtres (stone) ---

building_adastra_cave = {
	can_build = yes
	base_buildtime = 240
	category = unity
	icon = "building_adastra_cave"

	# Obligatoire en 4.4 : determine les zones ou le batiment peut etre eleve.
	building_sets = {
		urban_automation
		unity
	}

	prerequisites = { "tech_adastra_cave_art" }

	# Un seul par planete. ATTENTION : ne JAMAIS mettre cette limite
	# dans potential - un batiment dont le potential devient faux est
	# detruit par le moteur, donc il se construisait puis disparaissait
	# aussitot. planet_limit est le champ prevu pour ca (249 usages
	# dans le jeu de base).
	planet_limit = {
		base = 1
	}

	potential = {
		exists = owner
		owner = { has_origin = origin_adastra }
	}

	# Construction reservee au confinement ; l'existant survit a l'emergence.
	allow = {
		owner = { NOT = { has_country_flag = adastra_completed } }
	}

	resources = {
		category = planet_buildings
		cost = { minerals = 40 }
		upkeep = {
			energy = 1
			trigger = { owner = { adastra_pays_energy_upkeep = yes } }
		}
		upkeep = {
			minerals = 1
			food = 1
			trigger = { owner = { NOT = { adastra_pays_energy_upkeep = yes } } }
		}
	}

	# La production vient d'emplois, pas d'un pourcentage : le Grenier
	# emploie des fermiers. Script du jeu de base, qui gere deja les
	# variantes gestalt et empire dechu.
	inline_script = {
		script = jobs/unity_jobs_add
		AMOUNT = @building_static_jobs_low_3
	}

	planet_modifier = {
		planet_stability_add = 3
	}

	ai_weight = { weight = 5 }
}
