# Ad Astra
# Origine : commencer en tant que civilisation pré-PRL et gravir tous les âges jusqu'aux étoiles.

origin_adastra = {
	is_origin = yes
	icon = "gfx/interface/icons/origins/origin_default_pre_ftl.dds"

	# 1.2 : l'image d'Unification durable (une metropole qui s'eleve) plutot que
	# celle de Colonie perdue (une epave). Ad Astra ne raconte pas un naufrage :
	# c'est une civilisation qui se construit sur son propre monde.
	picture = GFX_evt_metropolis

	description = "origin_adastra_effects"

	playable = {
		always = yes
	}

	# 1.1 : réservée aux empires réguliers - le confinement gestalt (ruche/machine)
	# aura son contenu dédié plus tard, jouer l'origine en gestalt aujourd'hui casserait la partie.
	possible = {
		authority = {
			NOR = {
				value = auth_hive_mind
				value = auth_machine_intelligence
			}
		}
	}

	flags = { adastra_origin }

	advanced_start = no
	preferred_planet_class_neighbor = yes

	# 1.2 : l'IA peut prendre l'origine.
	#
	# Jusqu'en 1.1 le poids était à 0 : l'IA aurait été paralysée, faute de
	# personnalité valide sous le type de pays adastra_grounded et faute de
	# poids sur les décisions du programme spatial. Les deux manques sont
	# comblés (common/personalities/zzz_adastra_personalities.txt et les
	# ai_weight des décisions), donc l'origine s'ouvre à l'IA.
	#
	# Le poids reste volontairement bas : c'est une origine à contrainte forte,
	# une galaxie qui en compterait cinq n'aurait plus de grandes puissances au
	# départ. base = 1 la rend possible sans la rendre courante — pour une
	# galaxie entièrement pré-PRL, c'est le mode dédié qui viendra, pas le
	# tirage aléatoire.
	random_weight = {
		base = 1
	}
}
